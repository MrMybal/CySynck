cmake_minimum_required(VERSION 3.24)
project(CySynck VERSION 0.7.0 LANGUAGES C CXX)
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path/../lib")
elseif(UNIX)
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(CYSYNCK_DESKTOP "Build the native CyUI desktop" ON)
option(CYSYNCK_TESTS "Build integration tests" ON)
include(FetchContent)
find_package(Threads REQUIRED)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG 5814e121dfb5049f72a5c3956c3c8961b40da78b)
set(HTTPLIB_REQUIRE_OPENSSL OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(picosha GIT_REPOSITORY https://github.com/okdshin/PicoSHA2.git GIT_TAG 161cb3fc4170fa7a3eca9e582cebd27cc4d1fe29 SOURCE_SUBDIR _no_cmake)
FetchContent_MakeAvailable(json httplib)
FetchContent_MakeAvailable(picosha)
set(tinyxml2_BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(tinyxml2 GIT_REPOSITORY https://github.com/leethomason/tinyxml2.git GIT_TAG 321ea883b7190d4e85cae5512a12e5eaa8f8731f)
FetchContent_MakeAvailable(tinyxml2)
add_library(cysynck_core SHARED src/core.cpp src/backup.cpp src/diff.cpp src/resolution.cpp src/process.cpp src/child_process.cpp src/engine.cpp src/network.cpp src/http.cpp src/c_api.cpp src/commands.cpp)
set_target_properties(cysynck_core PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_definitions(cysynck_core PRIVATE CYSYNCK_BUILDING_CORE)
target_include_directories(cysynck_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ${picosha_SOURCE_DIR})
target_link_libraries(cysynck_core PUBLIC $<BUILD_INTERFACE:nlohmann_json::nlohmann_json> PRIVATE httplib::httplib tinyxml2::tinyxml2 Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(cysynck_core PROPERTIES EXPORT_NAME Core)
add_library(CySynck::Core ALIAS cysynck_core)
if(WIN32)
  target_link_libraries(cysynck_core PRIVATE bcrypt ws2_32 advapi32)
endif()
add_library(cysynck_startup STATIC src/startup.cpp)
target_include_directories(cysynck_startup PUBLIC src)
target_link_libraries(cysynck_startup PUBLIC cysynck_core)
if(WIN32)
  target_link_libraries(cysynck_startup PRIVATE shell32 ole32 uuid)
endif()
add_executable(cysynck src/cli.cpp)
target_link_libraries(cysynck PRIVATE cysynck_startup)
add_custom_command(TARGET cysynck POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/web" "$<TARGET_FILE_DIR:cysynck>/web")
if(CYSYNCK_DESKTOP)
  FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG 1ab953d75a760d5102eb2d5c4c6c4e65df6b29f3 SOURCE_SUBDIR _no_cmake)
  FetchContent_Declare(cyui GIT_REPOSITORY https://github.com/MrMybal/CyUI.git GIT_TAG a228501d9418550ce68d2d67b70cebbb8fa3ee2a)
  FetchContent_MakeAvailable(glfw)
  FetchContent_MakeAvailable(imgui)
  find_package(OpenGL REQUIRED)
  add_library(cysynck_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/misc/cpp/imgui_stdlib.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(cysynck_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(cysynck_imgui PUBLIC glfw OpenGL::GL)
  set(CYUI_IMGUI_TARGET cysynck_imgui)
  set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
  set(CYUI_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
  set(CYUI_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
  set(CYUI_MODULES "Diff;Inspector")
  FetchContent_MakeAvailable(cyui)
  add_library(cysynck_desktop_support STATIC src/desktop_support.cpp)
  target_link_libraries(cysynck_desktop_support PUBLIC cysynck_startup)
  target_include_directories(cysynck_desktop_support PUBLIC src)
  if(WIN32)
    target_sources(cysynck_desktop_support PRIVATE src/tray_windows.cpp)
    target_link_libraries(cysynck_desktop_support PRIVATE shell32 user32 gdi32)
  elseif(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set_property(TARGET cysynck_desktop_support PROPERTY OBJCXX_STANDARD 20)
    target_sources(cysynck_desktop_support PRIVATE src/tray_macos.mm)
    set_source_files_properties(src/tray_macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(cysynck_desktop_support PRIVATE "-framework AppKit")
  else()
    target_sources(cysynck_desktop_support PRIVATE src/tray_linux.cpp)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(APPINDICATOR QUIET IMPORTED_TARGET ayatana-appindicator3-0.1)
    endif()
    if(APPINDICATOR_FOUND)
      target_compile_definitions(cysynck_desktop_support PRIVATE CYSYNCK_APPINDICATOR)
      target_link_libraries(cysynck_desktop_support PRIVATE PkgConfig::APPINDICATOR)
    else()
      message(STATUS "Ayatana AppIndicator unavailable: desktop will stay visible when no tray is available")
    endif()
  endif()
  add_executable(cysynck-desktop src/desktop.cpp)
  target_link_libraries(cysynck-desktop PRIVATE cysynck_desktop_support CyUI::Diff CyUI::Inspector)
  if(WIN32)
    target_link_libraries(cysynck-desktop PRIVATE shell32)
    set_target_properties(cysynck-desktop PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
endif()
if(MSVC)
  add_compile_options(/utf-8 /W4 /EHsc)
  target_compile_options(cysynck_core PRIVATE /utf-8 /W4 /EHsc)
  target_compile_options(cysynck_startup PRIVATE /utf-8 /W4 /EHsc)
  target_compile_options(cysynck PRIVATE /utf-8 /W4 /EHsc)
  if(TARGET cysynck-desktop)
    target_compile_options(cysynck-desktop PRIVATE /utf-8 /W4 /EHsc)
    target_compile_options(cysynck_desktop_support PRIVATE /utf-8 /W4 /EHsc)
  endif()
endif()
if(CYSYNCK_TESTS)
  enable_testing()
  add_executable(cysynck-startup-tests tests/startup_tests.cpp)
  target_link_libraries(cysynck-startup-tests PRIVATE cysynck_startup tinyxml2::tinyxml2)
  if(UNIX AND NOT APPLE)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(STARTUP_GIO QUIET IMPORTED_TARGET gio-unix-2.0)
      if(STARTUP_GIO_FOUND)
        target_link_libraries(cysynck-startup-tests PRIVATE PkgConfig::STARTUP_GIO)
        target_compile_definitions(cysynck-startup-tests PRIVATE CYSYNCK_TEST_GIO)
      endif()
    endif()
  endif()
  add_test(NAME cysynck-startup COMMAND cysynck-startup-tests)
  add_executable(cysynck-backup-tests tests/backup_tests.cpp)
  target_link_libraries(cysynck-backup-tests PRIVATE cysynck_core)
  add_test(NAME cysynck-backups COMMAND cysynck-backup-tests)
  add_executable(cysynck-diff-tests tests/diff_tests.cpp)
  target_link_libraries(cysynck-diff-tests PRIVATE cysynck_core)
  add_test(NAME cysynck-diffs COMMAND cysynck-diff-tests)
  add_executable(cysynck-resolution-tests tests/resolution_tests.cpp)
  target_link_libraries(cysynck-resolution-tests PRIVATE cysynck_core)
  add_test(NAME cysynck-resolutions COMMAND cysynck-resolution-tests)
  add_executable(cysynck-tests tests/core_tests.cpp)
  target_link_libraries(cysynck-tests PRIVATE cysynck_core httplib::httplib)
  add_test(NAME cysynck-integration COMMAND cysynck-tests $<TARGET_FILE:cysynck-example-plugin>)
  add_library(cysynck-example-plugin SHARED examples/plugin.cpp)
  target_include_directories(cysynck-example-plugin PRIVATE include)
  set(CYSYNCK_TEST_SYNCTHING "" CACHE FILEPATH "Optional Syncthing executable for real peer integration tests")
  add_executable(cysynck-sync-tests tests/sync_tests.cpp)
  target_link_libraries(cysynck-sync-tests PRIVATE cysynck_core httplib::httplib)
  if(CYSYNCK_TEST_SYNCTHING)
    add_test(NAME cysynck-real-sync COMMAND cysynck-sync-tests "${CYSYNCK_TEST_SYNCTHING}")
    set_tests_properties(cysynck-real-sync PROPERTIES TIMEOUT 120)
  endif()
  if(CYSYNCK_DESKTOP)
    add_executable(cysynck-desktop-tests tests/desktop_tests.cpp)
    target_link_libraries(cysynck-desktop-tests PRIVATE cysynck_desktop_support)
    if(WIN32)
      target_link_libraries(cysynck-desktop-tests PRIVATE ws2_32)
    endif()
    add_test(NAME cysynck-desktop-session COMMAND cysynck-desktop-tests)
    if(WIN32 AND CYSYNCK_TEST_SYNCTHING)
      add_test(NAME cysynck-desktop-background COMMAND cysynck-desktop-tests $<TARGET_FILE:cysynck-desktop> "${CYSYNCK_TEST_SYNCTHING}")
      set_tests_properties(cysynck-desktop-background PROPERTIES TIMEOUT 90)
    endif()
    set(CYSYNCK_TEST_PACKAGE "" CACHE PATH "Optional installed package tree for Windows deployment tests")
    if(WIN32 AND CYSYNCK_TEST_PACKAGE)
      add_test(NAME cysynck-deployment COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/deployment_tests.ps1" -Package "${CYSYNCK_TEST_PACKAGE}")
      set_tests_properties(cysynck-deployment PROPERTIES TIMEOUT 120)
    endif()
  endif()
endif()
install(TARGETS cysynck RUNTIME DESTINATION bin)
install(TARGETS cysynck_core EXPORT CySynckTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(EXPORT CySynckTargets NAMESPACE CySynck:: DESTINATION lib/cmake/CySynck)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/CySynckConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/CySynckConfig.cmake" INSTALL_DESTINATION lib/cmake/CySynck)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/CySynckConfigVersion.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CySynckConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/CySynckConfigVersion.cmake" DESTINATION lib/cmake/CySynck)
if(TARGET cysynck-desktop)
  install(TARGETS cysynck-desktop RUNTIME DESTINATION bin)
endif()
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY "${json_SOURCE_DIR}/include/nlohmann" DESTINATION include)
install(DIRECTORY assets/web DESTINATION share/cysynck)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cysynck-version.json" "{\"version\":\"${PROJECT_VERSION}\"}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/cysynck-version.json" DESTINATION share/cysynck RENAME version.json)
if(WIN32 AND CYSYNCK_DESKTOP)
  install(FILES deploy/Install-CySynck.ps1 DESTINATION .)
endif()
install(FILES README.md README.fr.md THIRD_PARTY.md DESTINATION share/cysynck)
install(DIRECTORY docs/ DESTINATION share/cysynck/docs)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION share/cysynck/licenses RENAME nlohmann-json.txt)
install(FILES "${httplib_SOURCE_DIR}/LICENSE" DESTINATION share/cysynck/licenses RENAME cpp-httplib.txt)
install(FILES "${picosha_SOURCE_DIR}/LICENSE" DESTINATION share/cysynck/licenses RENAME PicoSHA2.txt)
install(FILES "${tinyxml2_SOURCE_DIR}/LICENSE.txt" DESTINATION share/cysynck/licenses RENAME TinyXML2.txt)
if(CYSYNCK_DESKTOP)
  install(FILES "${cyui_SOURCE_DIR}/LICENSE" DESTINATION share/cysynck/licenses RENAME CyUI.txt)
  install(FILES "${imgui_SOURCE_DIR}/LICENSE.txt" DESTINATION share/cysynck/licenses RENAME DearImGui.txt)
  install(FILES "${glfw_SOURCE_DIR}/LICENSE.md" DESTINATION share/cysynck/licenses RENAME GLFW.txt)
  install(FILES "${cyui_SOURCE_DIR}/third_party/lexilla/License.txt" DESTINATION share/cysynck/licenses RENAME Lexilla-Scintilla.txt)
  install(FILES "${cyui_SOURCE_DIR}/third_party/SciTE-Properties-License.txt" DESTINATION share/cysynck/licenses)
endif()
if(MSVC)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  include(InstallRequiredSystemLibraries)
endif()
set(CPACK_PACKAGE_NAME CySynck)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP)
include(CPack)
