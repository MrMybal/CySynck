cmake_minimum_required(VERSION 3.24)
project(CySynckReportPlugin LANGUAGES CXX)
find_package(CySynck 0.30 CONFIG REQUIRED)
add_library(cysynck-report-plugin SHARED plugin.cpp)
target_compile_features(cysynck-report-plugin PRIVATE cxx_std_20)
target_link_libraries(cysynck-report-plugin PRIVATE CySynck::PluginSDK)
if(MSVC)
  target_compile_options(cysynck-report-plugin PRIVATE /utf-8 /W4 /EHsc)
endif()
