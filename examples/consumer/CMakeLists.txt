cmake_minimum_required(VERSION 3.24)
project(CySynckConsumer LANGUAGES CXX)
find_package(CySynck 0.1 CONFIG REQUIRED)
add_executable(cysynck-consumer main.cpp)
target_compile_features(cysynck-consumer PRIVATE cxx_std_20)
target_link_libraries(cysynck-consumer PRIVATE CySynck::Core)
if(WIN32)
  add_custom_command(TARGET cysynck-consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:CySynck::Core>" "$<TARGET_FILE_DIR:cysynck-consumer>")
endif()
